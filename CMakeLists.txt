# Build identity is captured here rather than from the runner's current directory.
set(BETAGO_GIT_AVAILABLE false)
set(BETAGO_GIT_DIRTY false)
set(BETAGO_GIT_REVISION "")
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BETAGO_GIT_REVISION
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE BETAGO_GIT_RESULT)
    if(BETAGO_GIT_RESULT EQUAL 0)
        execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=normal
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BETAGO_GIT_STATUS
            OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE BETAGO_STATUS_RESULT)
        if(BETAGO_STATUS_RESULT EQUAL 0)
            set(BETAGO_GIT_AVAILABLE true)
            if(NOT BETAGO_GIT_STATUS STREQUAL "")
                set(BETAGO_GIT_DIRTY true)
            endif()
        endif()
    endif()
    set(BETAGO_GIT_DEPENDENCIES HEAD index packed-refs)
    execute_process(COMMAND "${GIT_EXECUTABLE}" symbolic-ref -q HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BETAGO_CURRENT_REF
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE BETAGO_REF_RESULT)
    if(BETAGO_REF_RESULT EQUAL 0)
        list(APPEND BETAGO_GIT_DEPENDENCIES "${BETAGO_CURRENT_REF}")
    endif()
    foreach(BETAGO_GIT_DEPENDENCY IN LISTS BETAGO_GIT_DEPENDENCIES)
        execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --git-path "${BETAGO_GIT_DEPENDENCY}"
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BETAGO_GIT_PATH
            OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE BETAGO_PATH_RESULT)
        if(BETAGO_PATH_RESULT EQUAL 0)
            get_filename_component(BETAGO_GIT_PATH "${BETAGO_GIT_PATH}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
            if(EXISTS "${BETAGO_GIT_PATH}")
                set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BETAGO_GIT_PATH}")
            endif()
        endif()
    endforeach()
endif()
file(GLOB_RECURSE BETAGO_SOURCE_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "include/*.hpp" "src/*.cpp" "tests/*.cpp")
list(APPEND BETAGO_SOURCE_FILES "build.ps1" "CMakeLists.txt" "cmake/build_info.hpp.in")
list(SORT BETAGO_SOURCE_FILES)
set(BETAGO_SOURCE_MANIFEST "")
foreach(BETAGO_SOURCE_FILE IN LISTS BETAGO_SOURCE_FILES)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/${BETAGO_SOURCE_FILE}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${BETAGO_SOURCE_FILE}" BETAGO_FILE_HASH)
    string(APPEND BETAGO_SOURCE_MANIFEST "${BETAGO_SOURCE_FILE}:${BETAGO_FILE_HASH}\n")
endforeach()
string(SHA256 BETAGO_SOURCE_SHA256 "${BETAGO_SOURCE_MANIFEST}")
string(TIMESTAMP BETAGO_BUILD_TIME "%Y-%m-%dT%H:%M:%SZ" UTC)
set(BETAGO_COMPILER "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
set(BETAGO_BUILD_FLAGS "${CMAKE_CXX_FLAGS}")
set(BETAGO_PROFILES Debug Release RelWithDebInfo MinSizeRel ${CMAKE_CONFIGURATION_TYPES} ${CMAKE_BUILD_TYPE})
list(REMOVE_DUPLICATES BETAGO_PROFILES)
foreach(BETAGO_PROFILE IN LISTS BETAGO_PROFILES)
    string(TOUPPER "${BETAGO_PROFILE}" BETAGO_PROFILE_UPPER)
    string(APPEND BETAGO_BUILD_FLAGS "$<$<CONFIG:${BETAGO_PROFILE}>: ${CMAKE_CXX_FLAGS_${BETAGO_PROFILE_UPPER}}>")
endforeach()
set(BETAGO_BUILD_PROFILE "${CMAKE_BUILD_TYPE}")
configure_file(cmake/build_info.hpp.in generated/build_info.config.hpp @ONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/build_info.hpp"
    INPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/build_info.config.hpp")
